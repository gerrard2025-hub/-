## fast api 代码

```python
import uvicorn
from fastapi import FastAPI,Request
from langchain.chat_models import init_chat_model
from langchain_openai import ChatOpenAI
import logging

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s %(levelname)s %(message)s",
)
logger = logging.getLogger(__name__)

logger.info("收到请求")
logger.warning("模型返回空内容")
logger.error("调用失败")

#=================================1.配置模型=================================
model = init_chat_model(
    model="qwen3.8-flash",
    base_url="https://cn-beijing.maas.qianwenaiapi.com/compatible-mode/v1",
    api_key="",
    max_tokens = 1000,
    temperature = 1.2,
    top_p = 0.9,
    model_provider="openai",
)

model2 = ChatOpenAI(
    model= "qwen2.5-coder:3b ",
    base_url="http://localhost:11434/v1",
    api_key="123",
    max_tokens=1024,
    temperature=1.2,
    top_p=0.9,
)
#=================================2.fast api处理=============================
app = FastAPI()
@app.post("/user")
async def chat(request: Request):
    #2.1读取请求体中内容
    data = await request.json()
    messages=data.get("messages")

    #2.2模型处理请求内容+容错机制
    #容错：返回内容为空，模型多次处理
    for i in range(3):
        response = model.invoke(messages)
        if response.content is not None:
            logger.info("使用模型1成功，重试次数为%d", i + 1)
            break

        #容错：500模型问题
        if response.status_code == 500:
            logger.error("模型网络错误，已经使用替换模型2")
            response = model2.invoke(messages)
            break

    return {"assistant":response.content}

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

## requests代码

```python
import requests
import logging

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s %(levelname)s %(message)s",
)
logger = logging.getLogger(__name__)

logger.info("收到请求")
logger.warning("模型返回空内容")
logger.error("调用失败")


base_url = "http://127.0.0.1:8000/user"

data = {
    "messages":[
        {"role":"system","content":"你是一名文学家，你擅长诗词歌赋，你的回答必须温文尔雅，用词生动"}
    ]
}

def multi_chat(data):
    while True:
        #构造请求体
        input_message = input("请输入你的问题")
        data["messages"].append({"role":"user","content":input_message})
        #连接fast api,返回响应结果
        post_resp = requests.post(url=base_url, json=data)
        #容错：502
        if post_resp.status_code == 502:
            logger.error("网关连接失败，请更换网址等信息")
            break
        #容错：504
        if post_resp.status_code == 504:
            logger.error("网关连接超时，请稍后重试")
            break
        #返回相应结果
        resoponse = post_resp.json()


        print(resoponse["assistant"])
        data["messages"].append({"role":"assistant","content":resoponse["assistant"]})
        print("-"*100)

if __name__ == "__main__":
    multi_chat(data)
```
