1. 写一个具有长期记忆且能压缩记忆的对话模型

   ```
   from langchain_community.chat_message_histories import RedisChatMessageHistory
   from langchain_core.prompts import ChatPromptTemplate,MessagesPlaceholder
   from langchain.chat_models import init_chat_model
   from langchain_core.runnables.history import RunnableWithMessageHistory
   from langchain_community.chat_message_histories import SQLChatMessageHistory
   from langchain_core.messages import SystemMessage,HumanMessage
   from sqlalchemy import create_engine
   
   
   
   #======================================1.构建模型========================================
   model = init_chat_model(
       model= "qwen2.5-coder:3b ",
       base_url="http://localhost:11434/v1",
       api_key="123",
       max_tokens=1000,
       temperature=1.2,
       top_p=0.9,
       model_provider="openai",
       extra_body={"enable_thinking": False})
   
   
   #=======================================2.构建长期记忆存储容器=================================
   # #方案一：用redis构建：
   # def get_session_history(session_id: str):
   #     return RedisChatMessageHistory(session_id,url="redis://192.168.2.197:6379/13")
   
   
   #方案二：用SQL构建
   #创建链接器：
   engine = create_engine("mysql+pymysql://root:")
   def get_session_history(session_id: str):
       return SQLChatMessageHistory(
           session_id=session_id,
           connection=engine, 
           table_name="chat_message_store"  # 可自定义表名，默认是 message_store
       )
   
   #=======================================3.构建对话模板 + 管道===================================
   #prompt 里用 MessagesPlaceholder 放历史消息
   prompt = ChatPromptTemplate.from_messages([
       ("system", "你是一个有帮助的AI助手"),
       MessagesPlaceholder(variable_name="history"),
       ("human", "{input}"),
   ])
   
   
   chain = prompt | model
   
   #==================================4.构造带记忆的链==================================
   #包装成带记忆的链  缠绕式
   chain_with_history = RunnableWithMessageHistory(
       chain,
       get_session_history,
       input_messages_key="input",
       history_messages_key="history",
   )
   
   #================================5.压缩信息功能=====================================
   def compress_and_store(session_id, llm, max_messages=20, compress_count=10):
       """压缩并存储消息：保留最近的 max_messages 条，将更早的 compress_count 条压缩为摘要"""
       history = get_session_history(session_id)
       messages = history.messages
       
       # 如果没超过限制，直接返回
       if len(messages) <= max_messages:
           return
           
       print(f"\n[系统提示] 消息已达 {len(messages)} 条，正在压缩早期记忆...")
       
       # 1. 切分消息：前 compress_count 条用于总结，剩下的保留
       # 注意：为了避免截断对话，最好保证 compress_count 是偶数（一问一答）
       old_messages = messages[:compress_count]
       keep_messages = messages[compress_count:]
       
       # 2. 构造 Prompt 让大模型总结这早期的对话
       history_text = ""
       for msg in old_messages:
           role = "用户" if isinstance(msg, HumanMessage) else "AI"
           history_text += f"{role}: {msg.content}\n"
           
       summary_prompt =    summary_prompt = f"""请将以下历史对话总结成一段简短的背景信息，保留用户的核心需求、身份信息和重要事实，忽略寒暄。字数控制在100字以内。
                                               【历史对话】：
                                               {history_text}
                                               【摘要】："""
                                               
       # 调用大模型生成摘要
       summary_response = llm.invoke(summary_prompt)
       summary_content = summary_response.content
       
       # 3. 清空数据库里的旧记录
       history.clear()
       
       # 4. 将摘要作为第一条系统消息写入
       summary_message = SystemMessage(content=f"[早期对话摘要]：{summary_content}")
       history.add_message(summary_message)
       
       # 5. 把保留下来的最近消息重新写回去
       history.add_messages(keep_messages)
       print(f"[系统提示] 压缩完成！已将前 {compress_count} 条消息浓缩为摘要，当前剩余 {len(history.messages)} 条。\n")
   
   
   def main():
       num = input("请输入客户id:")
       user_id = "user-" + str(num)
       while True:
           input_message = input("请输入你的问题：")
           if input_message == "exit":
               break
           else:
               res = chain_with_history.invoke({"input": input_message},config={"configurable": {"session_id": user_id}})
               print(f"ai:{res.content}")
       
       #压缩信息成为摘要
           compress_and_store(user_id,model)
           
   if __name__ == "__main__":
       main()
           
          
   ```
